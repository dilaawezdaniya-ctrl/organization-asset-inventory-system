CREATE DATABASE organization_inventory;

USE organization_inventory;

CREATE TABLE organization (
    organization_id INT AUTO_INCREMENT PRIMARY KEY,
    organization_name VARCHAR(150) NOT NULL,
    organization_type VARCHAR(100),
    address VARCHAR(255),
    contact_number VARCHAR(20),
    email VARCHAR(150)
);

CREATE TABLE department (
    department_id INT AUTO_INCREMENT PRIMARY KEY,
    organization_id INT NOT NULL,
    department_name VARCHAR(150) NOT NULL,
    department_head VARCHAR(150),

    UNIQUE (organization_id, department_name),

    FOREIGN KEY (organization_id)
        REFERENCES organization(organization_id)
);

CREATE TABLE category (
    category_id INT AUTO_INCREMENT PRIMARY KEY,
    category_name VARCHAR(100) NOT NULL,
    asset_type VARCHAR(100) NOT NULL,

    UNIQUE (category_name)
);

CREATE TABLE building (
    building_id INT AUTO_INCREMENT PRIMARY KEY,
    organization_id INT NOT NULL,
    building_name VARCHAR(150) NOT NULL,
    building_location VARCHAR(255),

    UNIQUE (organization_id, building_name),

    FOREIGN KEY (organization_id)
        REFERENCES organization(organization_id)
);

CREATE TABLE room (
    room_id INT AUTO_INCREMENT PRIMARY KEY,
    building_id INT NOT NULL,
    room_name VARCHAR(150) NOT NULL,
    room_type VARCHAR(100),

    UNIQUE (building_id, room_name),

    FOREIGN KEY (building_id)
        REFERENCES building(building_id)
);
CREATE TABLE asset (
    asset_id INT AUTO_INCREMENT PRIMARY KEY,
    asset_code VARCHAR(50) NOT NULL UNIQUE,
    asset_name VARCHAR(150) NOT NULL,
    serial_number VARCHAR(100),

    category_id INT NOT NULL,
    department_id INT NOT NULL,
    room_id INT NOT NULL,

    purchase_date DATE NOT NULL,
    purchase_price DECIMAL(12,2),

    `condition` VARCHAR(50) NOT NULL,
    status VARCHAR(50) NOT NULL,

    FOREIGN KEY (category_id)
        REFERENCES category(category_id),

    FOREIGN KEY (department_id)
        REFERENCES department(department_id),

    FOREIGN KEY (room_id)
        REFERENCES room(room_id)
);


CREATE TABLE computer_details (
    computer_id INT AUTO_INCREMENT PRIMARY KEY,
    asset_id INT NOT NULL UNIQUE,
    hostname VARCHAR(100),
    processor VARCHAR(150),
    ram VARCHAR(50),
    storage VARCHAR(100),
    operating_system VARCHAR(100),

    FOREIGN KEY (asset_id)
        REFERENCES asset(asset_id)
);

CREATE TABLE security_details (
    security_id INT AUTO_INCREMENT PRIMARY KEY,
    computer_id INT NOT NULL UNIQUE,

    antivirus_status VARCHAR(50),
    firewall_status VARCHAR(50),
    security_status VARCHAR(50),
    last_security_check DATE,

    FOREIGN KEY (computer_id)
        REFERENCES computer_details(computer_id)
);

CREATE TABLE supplier (
    supplier_id INT AUTO_INCREMENT PRIMARY KEY,
    supplier_name VARCHAR(150) NOT NULL,
    contact_person VARCHAR(150),
    contact_number VARCHAR(20),
    email VARCHAR(150),
    address VARCHAR(255)
);


CREATE TABLE purchase (
    purchase_id INT AUTO_INCREMENT PRIMARY KEY,
    supplier_id INT NOT NULL,
    purchase_date DATE NOT NULL,
    invoice_number VARCHAR(100) UNIQUE,
    total_amount DECIMAL(12,2),
    purchase_status VARCHAR(50) NOT NULL,

    FOREIGN KEY (supplier_id)
        REFERENCES supplier(supplier_id)
);

CREATE TABLE purchase_item (
    purchase_item_id INT AUTO_INCREMENT PRIMARY KEY,

    purchase_id INT NOT NULL,
    category_id INT NOT NULL,

    quantity INT NOT NULL,
    unit_price DECIMAL(12,2) NOT NULL,
    total_price DECIMAL(12,2) NOT NULL,

    FOREIGN KEY (purchase_id)
        REFERENCES purchase(purchase_id),

    FOREIGN KEY (category_id)
        REFERENCES category(category_id)
);

    purchase_date DATE NOT NULL,
    purchase_price DECIMAL(12,2),

    `condition` VARCHAR(50) NOT NULL,
    status VARCHAR(50) NOT NULL,

    purchase_item_id INT,

    FOREIGN KEY (category_id)
        REFERENCES category(category_id),

    FOREIGN KEY (department_id)
        REFERENCES department(department_id),

    FOREIGN KEY (room_id)
        REFERENCES room(room_id),

    FOREIGN KEY (purchase_item_id)
        REFERENCES purchase_item(purchase_item_id)
);
